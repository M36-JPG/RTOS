void Timer0_Init(void)
{

	TCCR0A |= (1 << WGM01);
	TCCR0A |= (1 << WGM00);
	//TCCR0B &= ~(1 << WGM02);
	
      TCCR0A |=(1 << COM0A1);

	OCR0A = 0;

	TCCR0B |= (1 << CS01) | (1 << CS00);

	// Enable global interrupts
	//sei();
}
