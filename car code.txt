 SemaphoreHandle_t xcarSem;

void CarTask(void *pv){
    int id = * (int *)pv;
    while(1){
        if(xSemaphoreTake(xcarSem , portMAX_DELAY)==pdTRUE) // semaphore take
        {

            printf("Car %d take order\n",id); // car in 

            vTaskDelay(pdMS_TO_TICKS (6000)); // order

              printf("Car %d left\n",id); // car out

            xSemaphoreGive(xcarSem); // semaphore give

             vTaskDelay(pdMS_TO_TICKS (1000)); // delay in the end of the while
        }
    }
}
